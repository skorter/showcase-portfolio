export default function InteractiveBackground() {
    const containerRef = useRef(null);

    const rawMouseX = useMotionValue(0);
    const rawMouseY = useMotionValue(0);

    const mouseX = useSpring(rawMouseX, { stiffness: 80, damping: 20, mass: 0.4 });
    const mouseY = useSpring(rawMouseY, { stiffness: 80, damping: 20, mass: 0.4 });

    useEffect(() => {
        const element = containerRef.current;
        if (!element) return;

        const onMouseMove = (event) => {
            const rect = element.getBoundingClientRect();
            const x = event.clientX - rect.left - rect.width / 2;
            const y = event.clientY - rect.top - rect.height / 2;
            rawMouseX.set(x);
            rawMouseY.set(y);
        };

        const onMouseLeave = () => {
            rawMouseX.set(0);
            rawMouseY.set(0);
        }

        element.addEventListener("mousemove", onMouseMove);
        element.addEventListener("mouseleave", onMouseLeave);
        return () => {
            element.removeEventListener("mousemove", onMouseMove);
            element.removeEventListener("mouseleave", onMouseLeave);
        };
        }, [rawMouseX, rawMouseY]);
        
        return (
            <div ref={containerRef} className={styles.background}>
                {stickers.map((sticker) => (
                    <Sticker key={sticker.id} sticker={sticker} mouseX={mouseX} mouseY={mouseY} />
                ))}
            </div>
        );
}

function Sticker({ sticker, mouseX, mouseY }) {
    const [hovered, setHovered] = useState(false);

    const hoverWiggleX = useSpring(0, {stiffness: 200, damping: 12});
    const hoverWiggleY = useSpring(0, {stiffness: 200, damping: 12});

    useEffect(() => {
        if (!hovered) {
            hoverWiggleX.set(0);
            hoverWiggleY.set(0);
            return;
        }

        let raf;
        const tick = () => {
            hoverWiggleX.set(Math.random() * 4 - 2);
            hoverWiggleY.set(Math.random() * 4 - 2);
            raf = requestAnimationFrame(tick);
        };
        tick();

        return () => cancelAnimationFrame(raf);
    }, [hovered, hoverWiggleX, hoverWiggleY]);

    const parallaxX = useTransform(mouseX, (x) => x * sticker.depth);
    const parallaxY = useTransform(mouseY, (y) => y * sticker.depth);

    const finalX = useTransform([parallaxX, hoverWiggleX], ([p, w]) => p + w);
    const finalY = useTransform([parallaxY, hoverWiggleY], ([p, w]) => p + w);


    return (
        <motion.img
            src={sticker.src}
            alt=""
            draggable="false"
            className={styles.sticker}
            style={{
                left: sticker.x,
                top: sticker.y,
                width: `${sticker.size}px`,
                rotate: sticker.rotation,
                x: finalX,
                y: finalY,
            }}
            whileHover={{
                scale: 1.08,
                rotate: sticker.rotation + 3,
            }}
            transition={{ type: "spring", stiffness: 220, damping: 18 }}
            onMouseEnter={() => {
  console.log("hover", sticker.id);
  setHovered(true);
}}
            onMouseLeave={() => setHovered(false)}
        />
    )

}
